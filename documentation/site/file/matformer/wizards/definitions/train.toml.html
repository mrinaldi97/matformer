id = "train"
title = "Train a model"
description = "Build a training configuration and queue it as a job."
category = "Training"
order = 10

[[steps]]
id = "mode"
title = "From scratch or from an existing model"
description = "Choose how the model starts. A fine-tune inherits the architecture and tokenizer of a checkpoint."
[[steps.fields]]
name = "mode"
kind = "radio"
label = "Approach"
default = "scratch"
required = true
options = [
  { value = "scratch", label = "Train from scratch", doc = "Initialize a new model." },
  { value = "finetune", label = "Fine-tune", doc = "Start from an existing checkpoint or HuggingFace model." },
]

[[steps]]
id = "kind"
title = "Kind of model"
description = "The kind selects the learning signal, the model variant and the default hyperparameters."
[[steps.fields]]
name = "model_kind"
kind = "radio"
label = "Model"
default = "autoregressive"
required = true
options = [
  { value = "autoregressive", label = "Autoregressive", doc = "Causal language model. Predicts the next token." },
  { value = "masked", label = "Masked", doc = "BERT-like model. Reconstructs masked tokens." },
  { value = "sequence_classification", label = "Sequence classification", doc = "Assigns one label to a sequence." },
  { value = "token_classification", label = "Token classification", doc = "Assigns one label to each token." },
  { value = "other", label = "Other", doc = "Another registered model variant." },
]
[[steps.fields]]
name = "model_variant"
kind = "text"
label = "Registered model variant"
doc = "The variant name from the registry, for example multiprobe."
when = { field = "model_kind", is = "other" }
required = true

[[steps]]
id = "model"
title = "Model"
description = "Essential architecture values. Advanced parameters keep the preset defaults and are expanded below."
generator = "config"
preset = "${mode}_${model_kind}"
sections = ["model", "tokenizer"]

[[steps]]
id = "masking"
title = "Masking"
description = "How tokens are hidden from a masked model."
when = { field = "model_kind", is = "masked" }
generator = "config"
preset = "${mode}_${model_kind}"
sections = ["masking"]

[[steps]]
id = "data"
title = "Data"
description = "The dataset the model trains on. Choose an MDAT view or a file-backed dataset."
generator = "config"
preset = "${mode}_${model_kind}"
sections = ["data"]
variant_by = { data = { field = "data_kind", map = { mdat = "mdat", files = "lazy" } } }
[[steps.fields]]
name = "data_kind"
kind = "radio"
label = "Source"
default = "files"
options = [
  { value = "files", label = "Files", doc = "JSONL, CSV, CoNLL or HuggingFace dataset read as a stream." },
  { value = "mdat", label = "MDAT", doc = "A pretokenized view of an MDAT container." },
]
[[steps.fields]]
name = "test_data_root"
kind = "path"
label = "Test data (optional)"
doc = "A second dataset used by validation. Without it validation reads the training dataset."
[[steps.fields]]
name = "test_file_format"
kind = "choice"
label = "Test format"
default = "jsonl"
options = ["jsonl", "csv", "conll", "conllu", "hf"]
[[steps.fields]]
name = "test_split"
kind = "text"
label = "Test split"
doc = "HuggingFace split name, used only when the test format is hf."

[[steps]]
id = "training"
title = "Training"
description = "Optimizer, schedule and checkpoint naming."
generator = "config"
preset = "${mode}_${model_kind}"
sections = ["training"]

[[steps]]
id = "validation"
title = "Validation"
description = "Tasks run during training. Each row can name a configured data entry or point at a file."
[[steps.fields]]
name = "validation_enabled"
kind = "bool"
label = "Run validation"
default = false
[[steps.fields]]
name = "validation_tasks"
kind = "tasks"
label = "Tasks"
when = { field = "validation_enabled", truthy = true }
columns = [
  { name = "type", kind = "choice", label = "Task", width = 16, options = ["evaluation", "static_model_analysis", "activation_analysis", "sandbox_train"], default = "evaluation" },
  { name = "name", kind = "text", label = "Name", width = 12, default = "" },
  { name = "data", kind = "path", label = "Data", width = 28, default = "", doc = "Configured data name or a file path. Empty uses the test split or training split." },
]
default = [{ type = "evaluation", name = "evaluation", data = "" }]
[[steps.fields]]
name = "validation_every_n_epochs"
kind = "int"
label = "Every n epochs"
default = 1
when = { field = "validation_enabled", truthy = true }
[[steps.fields]]
name = "validation_every_n_steps"
kind = "int"
label = "Every n steps"
default = 0
doc = "Zero disables step-based validation."
when = { field = "validation_enabled", truthy = true }
[[steps.fields]]
name = "validation_sandbox_configs"
kind = "paths"
label = "Sandbox configs"
doc = "YAML files used by sandbox_train rows."
default = []
when = { field = "validation_enabled", truthy = true }

[[steps]]
id = "launch"
title = "Launch"
description = "Machine options for the queued job."
[[steps.fields]]
name = "device"
kind = "choice"
label = "Accelerator"
default = "auto"
options = ["auto", "cuda", "cpu", "mps", "tpu"]
[[steps.fields]]
name = "gpus"
kind = "text"
label = "Pin to GPUs"
doc = "CUDA_VISIBLE_DEVICES for this job only, for example 2,3."
[[steps.fields]]
name = "save_model"
kind = "bool"
label = "Save the model"
default = true
[[steps.fields]]
name = "save_plugins"
kind = "bool"
label = "Save the plugins"
default = false

[action]
kind = "train"
name = "${training.run_name}"
project = "${training.project_name}"
options = { device = "${device}", gpus = "${gpus}", save_model = "${save_model}", save_plugins = "${save_plugins}" }
