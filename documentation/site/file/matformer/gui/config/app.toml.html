# The shell is declared here: pages, menus, toolbar, themes, servers. No code changes needed.
# The Tasks group is generated from the local registry.

# Either a plain url (ssh -L tunnel or LAN), or a full [server.ssh] block: the console then
# probes, tunnels, and starts `python -m matformer.server` remotely if nothing answers.

  { name = "local", url = "127.0.0.1:8765" },
]
