"""
Publish a Matformer checkpoint to a HuggingFace repository.

The payload built here makes a Matformer repo usable with the usual HF tutorials:

    config.json              light config (model_type "matformer") + auto_map
    modeling_matformer.py    remote-code shim importing matformer.loading.huggingface_integration
    matformer.yaml           the real Matformer configuration
    model.safetensors        weights in Matformer's stable-state-dict format
    README.md                model card with copy-paste usage examples
    tokenizer files          (optional) copied from the checkpoint's tokenizer

Nothing needs to be loaded into a model to publish: the checkpoint is read
through the Matformer checkpoint holder, so any source the framework can read
(.ckpt, weights + YAML folder, legacy Lightning checkpoints) can be published.

Library use:

    from matformer.loading.huggingface_publish import push_to_hub
    push_to_hub("/path/to/model.ckpt", "user/model", token="hf_...")

CLI:

    python -m matformer.loading.huggingface_publish \
        --checkpoint /path/to/model.ckpt --repo_id user/model [--token hf_...]

    # build the payload locally without pushing (no credentials needed):
    python -m matformer.loading.huggingface_publish \
        --checkpoint /path/to/model.ckpt --out ./hf_payload
"""
import json
import shutil
import tempfile
import warnings
from pathlib import Path

import torch

from matformer.loading.huggingface_integration import (
    MATFORMER_CONFIG_FILENAME,
    MODELING_FILENAME,
    PUBLISH_EXCLUDED_SECTIONS,
    WEIGHTS_FILENAME,
    MatformerConfig,
    _auto_map,
    save_matformer_yaml,
)

MODELING_GIT_URL = "git+https://github.com/mrinaldi97/matformer.git"
TOKENIZER_FILE_PATTERNS = [
    "tokenizer.json", "tokenizer_config.json", "special_tokens_map.json",
    "added_tokens.json", "vocab.txt", "vocab.json", "merges.txt",
    "spiece.model", "sentencepiece.bpe.model", "tokenizer.model",
    "chat_template.jinja",
]


def detect_model_type(configs):
    """Infer the HF task ('causal', 'masked', 'sequence', 'token') from Matformer configs."""
    model = configs.model
    if hasattr(model, "scope"):
        return "token" if model.scope == "token" else "sequence"
    model_class = (getattr(model, "model_class", None) or "").lower()
    objective = (getattr(model, "training_objective", None) or "").lower()
    if "autoregressive" in model_class or objective in ("autoregressive", "causal", "next_token", "clm"):
        return "causal"
    if "bert" in model_class or objective in ("masked", "mlm"):
        return "masked"
    if getattr(model, "is_causal", None) is False:
        return "masked"
    return "causal"


def create_modeling_file():
    """Content of the remote-code shim pushed into the HF repo."""
    return f'''# {MODELING_FILENAME}
# Auto-generated by Matformer: lets HuggingFace load this model with
# trust_remote_code=True. It only re-exports Matformer's own HF classes.
import os
import sys

matformer_root = os.getenv("MATFORMER_ROOT")
if matformer_root:
    matformer_root = os.path.abspath(os.path.expanduser(matformer_root))
    if matformer_root not in sys.path:
        sys.path.insert(0, matformer_root)

try:
    from matformer.loading.huggingface_integration import (
        MatformerConfig,
        MatformerModel,
        MatformerForCausalLM,
        MatformerForMaskedLM,
        MatformerForSequenceClassification,
        MatformerForTokenClassification,
        register_matformer,
    )
except ImportError as import_error:
    raise ImportError(
        "This model requires Matformer. Install it with:\\n"
        f"  pip install {MODELING_GIT_URL}\\n"
        "or point the MATFORMER_ROOT environment variable to a local checkout."
    ) from import_error

register_matformer()
'''


def create_readme(repo_id, model_type, tokenizer_name=None):
    """Model card with a ready-to-run example for the detected task."""
    tokenizer_line = ""
    tokenizer_load = ""
    if tokenizer_name:
        tokenizer_load = 'tokenizer = AutoTokenizer.from_pretrained("%s")\n' % repo_id
        tokenizer_line = ("\n> If the repository has no tokenizer files, load the original\n"
                          f"> one instead: `AutoTokenizer.from_pretrained(\"{tokenizer_name}\")`.\n")
    examples = {
        "causal": f'''import torch
from transformers import AutoModelForCausalLM, AutoTokenizer

model = AutoModelForCausalLM.from_pretrained("{repo_id}", trust_remote_code=True)
{tokenizer_load}prompt = "Il transformer è un modello"
inputs = tokenizer(prompt, return_tensors="pt").to(model.device)
with torch.no_grad():
    out = model.generate(**inputs, max_new_tokens=50, do_sample=True, temperature=0.8)
print(tokenizer.decode(out[0], skip_special_tokens=True))''',
        "masked": f'''import torch
from transformers import AutoModelForMaskedLM, AutoTokenizer

model = AutoModelForMaskedLM.from_pretrained("{repo_id}", trust_remote_code=True)
{tokenizer_load}text = "Roma è la [MASK] d'Italia."
inputs = tokenizer(text, return_tensors="pt").to(model.device)
with torch.no_grad():
    logits = model(**inputs).logits
mask_pos = (inputs["input_ids"][0] == tokenizer.mask_token_id).nonzero()[0]
best = logits[0, mask_pos].argmax()
print(tokenizer.convert_ids_to_tokens(best))''',
        "sequence": f'''import torch
from transformers import AutoModelForSequenceClassification, AutoTokenizer

model = AutoModelForSequenceClassification.from_pretrained("{repo_id}", trust_remote_code=True)
{tokenizer_load}inputs = tokenizer("Un film bellissimo!", return_tensors="pt").to(model.device)
with torch.no_grad():
    print(model(**inputs).logits)''',
        "token": f'''import torch
from transformers import AutoModelForTokenClassification, AutoTokenizer

model = AutoModelForTokenClassification.from_pretrained("{repo_id}", trust_remote_code=True)
{tokenizer_load}inputs = tokenizer("Maria Rossi vive a Roma", return_tensors="pt").to(model.device)
with torch.no_grad():
    print(model(**inputs).logits.shape)''',
    }
    return f'''---
tags:
- matformer
library_name: transformers
---

# {repo_id.rsplit("/", 1)[-1] if "/" in repo_id else repo_id}

Trained with [Matformer](https://github.com/mrinaldi97/matformer) and published as a
HuggingFace-compatible model. The architecture, weights and generation logic all live
in Matformer: this repo only wraps them for the HF ecosystem.

## Installation

```bash
pip install {MODELING_GIT_URL}
```

## Usage

```python
{examples[model_type]}
```
{tokenizer_line}
## Notes

- `from_pretrained` accepts `device=` and `dtype=` (defaults: auto device, `torch.bfloat16`).
- Generation is Matformer's own sampling loop: no KV-cache, no beam search.
- Fine-tune on a single GPU (or via `accelerate`): multi-GPU `nn.DataParallel` is not
  supported by Matformer internals.
- Every Matformer checkpoint can be re-purposed, e.g.
  `AutoModelForSequenceClassification.from_pretrained("{repo_id}", num_labels=5, trust_remote_code=True)`
  attaches a fresh (Matformer) classification head on top of the loaded backbone.
'''


def _export_weights(raw_weights):
    """Tensors-only, CPU, deduplicated state dict, ready for safetensors.

    Old Lightning checkpoints stored every tensor twice (`model.x` alongside `x`):
    the prefixed duplicates are dropped. Tied weights share storage and are
    written once.
    """
    keys = set(raw_weights.keys())
    exported, seen = {}, set()
    for key, value in raw_weights.items():
        if not torch.is_tensor(value):
            continue
        if key.startswith("model.") and key[len("model."):] in keys:
            continue
        if value.data_ptr() in seen:
            continue
        seen.add(value.data_ptr())
        exported[key] = value.detach().cpu().contiguous()
    if not exported:
        raise ValueError("No tensors found in the checkpoint weights.")
    return exported


def copy_tokenizer_files(configs, output_dir):
    """Copy the tokenizer declared in the configs into the payload so that
    AutoTokenizer.from_pretrained(repo) works out of the box."""
    name = getattr(getattr(configs, "tokenizer", None), "pretrained_name", None)
    if not name:
        warnings.warn("Configs declare no tokenizer: the repo will not provide one.")
        return False
    source = Path(name)
    if not source.is_dir():
        from huggingface_hub import snapshot_download
        try:
            source = Path(snapshot_download(name, allow_patterns=TOKENIZER_FILE_PATTERNS))
        except Exception as error:  # noqa: BLE001 - publishing must not die on tokenizer issues
            warnings.warn(f"Could not fetch tokenizer '{name}' ({error}): "
                          "the repo will not provide one.")
            return False
    copied = False
    for pattern in TOKENIZER_FILE_PATTERNS:
        for f in source.glob(pattern):
            shutil.copy(f, Path(output_dir) / f.name)
            copied = True
    if not copied:
        warnings.warn(f"No tokenizer files found in {source}.")
    return copied


def build_hub_payload(checkpoint, output_dir, configs=None, model_type="auto",
                      include_tokenizer=True, clean=True, repo_id=None):
    """Build a HuggingFace-ready folder from any Matformer-readable checkpoint.

    Returns the output_dir. With clean=True the weights are re-exported as
    model.safetensors (no optimizer state, no pickles); otherwise the original
    checkpoint file is copied as-is.
    """
    from safetensors.torch import save_file
    from matformer.loading.utils import make_holder

    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    holder = make_holder(str(checkpoint), map_location="cpu")
    configs = configs if configs is not None else holder.get_configs()
    if configs is None or not hasattr(configs, "model"):
        raise ValueError(f"{checkpoint} carries no Matformer configs; pass configs=... "
                         "or use a checkpoint saved by Matformer.")
    if model_type == "auto":
        model_type = detect_model_type(configs)
    if model_type not in ("causal", "masked", "sequence", "token"):
        raise ValueError(f"Unknown model_type '{model_type}'.")

    if clean:
        save_file(_export_weights(dict(holder.get_weights())),
                  str(output_dir / WEIGHTS_FILENAME))
        checkpoint_file = WEIGHTS_FILENAME
    else:
        checkpoint_file = Path(checkpoint).name
        shutil.copy(checkpoint, output_dir / checkpoint_file)

    save_matformer_yaml(configs, output_dir / MATFORMER_CONFIG_FILENAME)

    hf_config = MatformerConfig(
        matformer_config_file=MATFORMER_CONFIG_FILENAME,
        checkpoint_file=checkpoint_file,
        auto_map=_auto_map([model_type]),
    ).sync_from_matformer(configs)
    # Manual write: PretrainedConfig.save_pretrained would copy/rewrite auto_map toward the
    # config class's defining module (see integration save_pretrained notes).
    (output_dir / "config.json").write_text(
        json.dumps(hf_config.to_dict(), indent=2, sort_keys=True) + "\n")

    (output_dir / MODELING_FILENAME).write_text(create_modeling_file())
    (output_dir / "README.md").write_text(
        create_readme(repo_id or hf_config.matformer_model_name or "matformer-model",
                      model_type,
                      tokenizer_name=getattr(
                          getattr(configs, "tokenizer", None), "pretrained_name", None)))
    if include_tokenizer:
        copy_tokenizer_files(configs, output_dir)
    return output_dir


def push_to_hub(checkpoint, repo_id, token=None, private=False, model_type="auto",
                configs=None, include_tokenizer=True, clean=True, commit_message=None):
    """Push a Matformer checkpoint to the hub as a HF-compatible model repo."""
    from huggingface_hub import HfApi, create_repo

    with tempfile.TemporaryDirectory(prefix="matformer_hub_") as tmp:
        build_hub_payload(checkpoint, tmp, configs=configs, model_type=model_type,
                          include_tokenizer=include_tokenizer, clean=clean, repo_id=repo_id)
        create_repo(repo_id, token=token, private=private, exist_ok=True)
        HfApi(token=token).upload_folder(
            folder_path=tmp, repo_id=repo_id, token=token,
            commit_message=commit_message or "Upload Matformer model")
    print(f"Model pushed to https://huggingface.co/{repo_id}")
    return repo_id


def upload_to_huggingface(checkpoint, config=None, hf_name=None, hf_token=None,
                          weights_only=True, model_type="auto", **ignored):
    """Deprecated alias of push_to_hub (old integration signature)."""
    if hf_name is None:
        raise ValueError("hf_name (the repo id) is required.")
    configs = None
    if config:
        from matformer.matformer_configs import MatformerConfigs
        configs = MatformerConfigs.create_configs(config) if isinstance(config, (str, Path)) else config
    return push_to_hub(checkpoint, hf_name, token=hf_token, model_type=model_type,
                       configs=configs, clean=weights_only)


def main():
    import argparse

    parser = argparse.ArgumentParser(
        description="Publish a Matformer checkpoint to HuggingFace.")
    parser.add_argument("--checkpoint", required=True,
                        help="Matformer checkpoint (.ckpt / weights+yaml folder)")
    parser.add_argument("--repo_id", default=None, help="Target hub repo, e.g. user/model")
    parser.add_argument("--token", default=None, help="HuggingFace token (or HF_TOKEN env)")
    parser.add_argument("--out", default=None,
                        help="Only build the payload into this local folder, don't push")
    parser.add_argument("--config", default=None,
                        help="Extra YAML/JSON Matformer config merged over the checkpoint's")
    parser.add_argument("--model_type", default="auto",
                        choices=["auto", "causal", "masked", "sequence", "token"])
    parser.add_argument("--no_tokenizer", action="store_true",
                        help="Don't copy tokenizer files into the repo")
    parser.add_argument("--keep_checkpoint", action="store_true",
                        help="Push the original .ckpt instead of a clean safetensors export")
    parser.add_argument("--private", action="store_true")
    args = parser.parse_args()

    configs = None
    if args.config:
        from matformer.matformer_configs import MatformerConfigs
        configs = MatformerConfigs.create_configs(args.config)

    if args.out:
        build_hub_payload(args.checkpoint, args.out, configs=configs,
                          model_type=args.model_type,
                          include_tokenizer=not args.no_tokenizer,
                          clean=not args.keep_checkpoint)
        print(f"Payload built in {args.out}")
        return
    if not args.repo_id:
        parser.error("--repo_id is required unless --out is used")
    push_to_hub(args.checkpoint, args.repo_id, token=args.token, private=args.private,
                model_type=args.model_type, configs=configs,
                include_tokenizer=not args.no_tokenizer, clean=not args.keep_checkpoint)


if __name__ == "__main__":
    main()
