# matformer/server/runs.py
"""Access to the MatformerLogger files: list the runs stored under the configured roots, fetch
their hyperparameters and metric series, and tail an active run, pushing every new row to the
connected GUIs as `metric` events (same message shape for history and live points, so the GUI
plot code has exactly one intake)."""
import json
import threading
from pathlib import Path

from matformer.server import config
from matformer.server.core import broadcast, rpc

TAILERS = {}                       # run_id -> Tailer
LOCK = threading.Lock()


def run_dirs() -> dict:
    """run_id -> directory, for every JsonlLogger run folder found under the run roots."""
    out = {}
    for root in config()["run_roots"]:
        for meta in Path(root).expanduser().glob("**/logs/*/meta.json"):
            out[meta.parent.name] = meta.parent
    return out


def _read_json(path: Path, default=None):
    try:
        return json.loads(path.read_text("utf-8"))
    except (OSError, json.JSONDecodeError):
        return default


@rpc("runs.list")
def runs_list(conn, p):
    """One row per run: id, project, run, version, host, started, last step. A run that is one
    point of a sweep also says which, and what that point changed."""
    rows = []
    for run_id, folder in sorted(run_dirs().items()):
        meta, config_json = _read_json(folder / "meta.json", {}), _read_json(folder / "config.json", {})
        rows.append({"id": run_id, "project": meta.get("project", "?"), "run": meta.get("run", run_id),
                     "version": meta.get("version"), "host": meta.get("host", ""),
                     "started": meta.get("started", ""), "dir": str(folder),
                     "sweep": (meta.get("sweep") or {}).get("name"),
                     "total_steps": config_json.get("training/total_steps"),
                     "steps": _last_step(folder)})
    return rows


def _last_step(folder: Path):
    last = None
    try:
        with open(folder / "run.jsonl", "rb") as fh:
            fh.seek(max(0, fh.seek(0, 2) - 4096))
            for line in fh:
                row = _parse(line)
                if row and row.get("step") is not None:
                    last = row["step"]
    except OSError:
        pass
    return last


def _parse(line):
    try:
        return json.loads(line)
    except json.JSONDecodeError:
        return None


@rpc("runs.config")
def runs_config(conn, p):
    """The flattened hyperparameters of a run (what wandb showed on its Config tab)."""
    return _read_json(run_dirs()[p["run"]] / "config.json", {})


@rpc("runs.metrics")
def runs_metrics(conn, p):
    """Full history: {key: [[step, value], ...]} read from run.jsonl. Rows whose key carries its
    own '<prefix>/step' (sandboxed loggers) are plotted against it."""
    return _series(run_dirs()[p["run"]] / "run.jsonl")


def _series(file: Path, since=0) -> dict:
    out = {}
    if not file.exists():
        return out
    with open(file, "rb") as fh:
        fh.seek(since)
        for line in fh:
            row = _parse(line)
            if not row:
                continue
            x = row.get("step")
            for key, value in (row.get("m") or {}).items():
                if isinstance(value, (int, float)) and not isinstance(value, bool):
                    if key.endswith("/step"):
                        x = value                       # a scoped logger brings its own axis
                    out.setdefault(key, []).append([x, value])
    return out


@rpc("runs.watch")
def runs_watch(conn, p):
    """Start tailing a run: every new row is pushed as metric {run, key, points} events;
    the current history is sent first, exactly like jobs.watch does."""
    run = p["run"]
    folder = run_dirs()[run]
    for key, points in _series(folder / "run.jsonl").items():
        conn.push("metric", {"run": run, "key": key, "points": points})
    with LOCK:
        if run not in TAILERS:
            TAILERS[run] = Tailer(run, folder).start()


@rpc("runs.unwatch")
def runs_unwatch(conn, p):
    with LOCK:
        tailer = TAILERS.pop(p["run"], None)
    if tailer:
        tailer.stop_flag.set()


class Tailer(threading.Thread):
    """Polls one run.jsonl and broadcasts the rows appended since the last sweep."""

    def __init__(self, run_id, folder: Path):
        super().__init__(name=f"tail-{run_id}", daemon=True)
        self.run, self.file = run_id, folder / "run.jsonl"
        self.pos, self.stop_flag = self.file.stat().st_size if self.file.exists() else 0, threading.Event()

    def run(self):
        while not self.stop_flag.wait(0.5):
            try:
                size = self.file.stat().st_size
            except OSError:
                continue
            if size < self.pos:                       # truncated (new run reused the folder)
                self.pos = 0
            if size == self.pos:
                continue
            with open(self.file, "rb") as fh:
                fh.seek(self.pos)
                rows = [row for line in fh if (row := _parse(line))]
                self.pos = fh.tell()
            for row in rows:
                x = row.get("step")
                for key, value in (row.get("m") or {}).items():
                    if isinstance(value, (int, float)) and not isinstance(value, bool):
                        broadcast("metric", {"run": self.run, "key": key, "points": [[x, value]],
                                             "time": row.get("t")})


# ── shared run catalog (sqlite): each server caches the whole fleet's history so offline
# laptops and cross-server tooling can answer "what runs exist, and on which box" ──────────────
@rpc("catalog.merge")
def catalog_merge(conn, p):
    """Adopt another machine's list of one server's runs: that server's rows are replaced."""
    from matformer.utils.catalog import Catalog
    return {"merged": Catalog().replace_server(p["server"], p["rows"])}


@rpc("catalog.upsert")
def catalog_upsert(conn, p):
    """Absorb a catalog snapshot: every run the caller knows about, each under the server it was
    seen on. Additive — a machine never forgets a run just because another one lacks it."""
    from matformer.utils.catalog import Catalog
    return {"merged": Catalog().upsert(p["rows"])}


@rpc("catalog.list")
def catalog_list(conn, p):
    from matformer.utils.catalog import Catalog
    return Catalog().all()
